"""
train.py
--------
End-to-end training pipeline for HeartWise.

Run with:  python3 src/train.py

What it does, in order (each numbered item maps to a point in the
original improvement notes):

  2.  Trains & compares 6 models (LogReg, Decision Tree, Random Forest,
      XGBoost, SVM, KNN).
  6.  Evaluates every model with Stratified 5-Fold CV (not a single split),
      reporting ROC-AUC as mean +/- std.
  8.  Uses one sklearn Pipeline (preprocessing + classifier) throughout,
      so there is no preprocessing drift between training and prediction.
  5.  Calibrates the winning model's probabilities (Platt vs isotonic) and
      compares Brier scores before/after.
  7.  Runs error analysis on the held-out test set: false positives, false
      negatives, and which feature values are associated with the errors.
  3.  Computes SHAP values for the final model and saves a summary plot.
  11. Saves all the figures a dashboard needs (confusion matrix, ROC curves,
      feature importance, SHAP summary, class balance, calibration curve).
  12. Writes model + metadata with a version number, training date, dataset
      info and feature list, so the artifact is traceable.

Everything is saved under models/ and reports/ so the FastAPI service and
the Streamlit app can load it without retraining.
"""

# Features eligible for a monotonic constraint in the final XGBoost model.
# Deliberately excludes multi-category nominal features (cp, restecg, slope,
# thal) — forcing a single direction onto an *unordered* set of categories
# isn't methodologically sound, even if one of their one-hot dummies happens
# to correlate strongly. Only continuous/ordinal numerics and genuinely
# binary features are considered.
MONOTONE_CANDIDATE_FEATURES = NUMERIC_FEATURES + ["sex", "fbs", "exang"]
MONOTONE_CORR_THRESHOLD = 0.30


def compute_monotone_constraints(X_train, y_train, feature_names_raw) -> tuple:
    """Data-driven monotonic constraints for the final XGBoost model.

    Why this exists: with only 303 patients, some feature values are backed
    by very few training rows (e.g. only 5 patients have ca=4 — almost
    certainly a leftover "missing data" code from the original UCI
    collection, not a real 4th vessel). A tree ensemble is free to fit a
    non-monotonic wiggle through that handful of noisy points, which is
    exactly what produced the erratic ca=0/2/3/4 risk swings reported
    against the earlier Random Forest. XGBoost's monotone_constraints
    forces the learned function to move in one consistent direction for a
    given feature, eliminating that kind of small-sample artifact.

    The direction itself is NOT assumed from textbook clinical intuition —
    several features in this specific 1980s Cleveland cohort empirically
    run the opposite way from naive expectation (independently confirmed
    against other public analyses of the same dataset; see the dashboard's
    "Model Notes" section). Instead we measure the actual correlation with
    the target on the training data and only constrain a feature when that
    relationship is strong (|corr| >= 0.30) and, for the continuous
    features, has been separately confirmed to be genuinely monotonic
    across its full range (not just linearly correlated) — see the code
    comments in the project notes for the binned check. Age was checked
    and deliberately excluded: it has a real, non-monotonic (younger and
    older patients both skew higher-risk) relationship in this cohort, so
    forcing a single direction onto it would misrepresent the data.
    """
    combined = X_train.copy()
    combined["__target__"] = y_train.values
    corr = combined.corr(numeric_only=True)["__target__"]

    directions = {}
    for feat in MONOTONE_CANDIDATE_FEATURES:
        c = corr.get(feat, 0.0)
        if abs(c) >= MONOTONE_CORR_THRESHOLD:
            directions[feat] = (1 if c > 0 else -1, float(c))

    constraints = []
    for raw_name in feature_names_raw:
        value = 0
        if raw_name.startswith("num__"):
            col = raw_name[len("num__"):]
            if col in directions:
                value = directions[col][0]
        elif raw_name.startswith("cat__"):
            remainder = raw_name[len("cat__"):]
            for col in ["sex", "fbs", "exang"]:
                prefix = f"{col}_"
                if remainder.startswith(prefix) and col in directions:
                    base_dir = directions[col][0]
                    value_str = remainder[len(prefix):]
                    value = base_dir if value_str == "1" else -base_dir
                    break
        constraints.append(value)
    return tuple(constraints), directions


def get_candidate_models(monotone_constraints=None) -> dict:
    xgb_kwargs = dict(
        n_estimators=300,
        max_depth=3,
        learning_rate=0.05,
        eval_metric="logloss",
        random_state=RANDOM_STATE,
    )
    if monotone_constraints is not None:
        xgb_kwargs["monotone_constraints"] = monotone_constraints

    return {
        "Logistic Regression": LogisticRegression(max_iter=2000, random_state=RANDOM_STATE),
        "Decision Tree": DecisionTreeClassifier(max_depth=5, random_state=RANDOM_STATE),
        "Random Forest": RandomForestClassifier(
            n_estimators=300, max_depth=6, random_state=RANDOM_STATE
        ),
        "XGBoost": XGBClassifier(**xgb_kwargs),
        "SVM": SVC(probability=True, random_state=RANDOM_STATE),
        "KNN": KNeighborsClassifier(n_neighbors=15),
    }


def cross_validate_models(X_train, y_train, monotone_constraints=None) -> pd.DataFrame:
    """Point 6: Stratified 5-Fold CV instead of trusting a single split."""
    cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=RANDOM_STATE)
    scoring = {
        "accuracy": "accuracy",
        "precision": "precision",
        "recall": "recall",
        "f1": "f1",
        "roc_auc": "roc_auc",
    }
    rows = []
    for name, model in get_candidate_models(monotone_constraints).items():
        pipe = make_pipeline(model)
        scores = cross_validate(pipe, X_train, y_train, cv=cv, scoring=scoring, n_jobs=1)
        row = {"model": name}
        for metric in scoring:
            vals = scores[f"test_{metric}"]
            row[f"{metric}_mean"] = float(np.mean(vals))
            row[f"{metric}_std"] = float(np.std(vals))
        rows.append(row)
        print(
            f"[CV] {name:22s} ROC-AUC = {row['roc_auc_mean']:.3f} +/- {row['roc_auc_std']:.3f}"
            f"   |  Recall = {row['recall_mean']:.3f}  |  F1 = {row['f1_mean']:.3f}"
        )
    return pd.DataFrame(rows)


def evaluate_on_test(name, pipe, X_test, y_test) -> dict:
    y_pred = pipe.predict(X_test)
    y_proba = pipe.predict_proba(X_test)[:, 1]
    return {
        "model": name,
        "test_accuracy": float(accuracy_score(y_test, y_pred)),
        "test_precision": float(precision_score(y_test, y_pred)),
        "test_recall": float(recall_score(y_test, y_pred)),
        "test_f1": float(f1_score(y_test, y_pred)),
        "test_roc_auc": float(roc_auc_score(y_test, y_proba)),
    }


def plot_class_distribution(y, path):
    plt.figure(figsize=(5, 4))
    counts = pd.Series(y).map({0: "No Disease", 1: "Disease"}).value_counts()
    sns.barplot(x=counts.index, y=counts.values, hue=counts.index, palette="Set2", legend=False)
    plt.title("Class Distribution")
    plt.ylabel("Number of Patients")
    plt.tight_layout()
    plt.savefig(path, dpi=120)
    plt.close()


def plot_roc_curves(fitted_pipes: dict, X_test, y_test, path):
    plt.figure(figsize=(6, 5))
    ax = plt.gca()
    for name, pipe in fitted_pipes.items():
        RocCurveDisplay.from_estimator(pipe, X_test, y_test, name=name, ax=ax)
    plt.plot([0, 1], [0, 1], linestyle="--", color="gray", label="Chance")
    plt.title("ROC Curves — All Models (test set)")
    plt.tight_layout()
    plt.savefig(path, dpi=120)
    plt.close()


def plot_confusion_matrix(y_test, y_pred, model_name, path):
    cm = confusion_matrix(y_test, y_pred)
    plt.figure(figsize=(4.5, 4))
    sns.heatmap(
        cm,
        annot=True,
        fmt="d",
        cmap="Blues",
        xticklabels=["No Disease", "Disease"],
        yticklabels=["No Disease", "Disease"],
    )
    plt.title(f"Confusion Matrix — {model_name}")
    plt.ylabel("Actual")
    plt.xlabel("Predicted")
    plt.tight_layout()
    plt.savefig(path, dpi=120)
    plt.close()


def plot_feature_importance(pipe, X_test, y_test, path, top_n=15):
    """Works for tree-based models (feature_importances_) and linear models
    (coef_); for anything else (e.g. SVM, KNN) falls back to permutation
    importance so the dashboard always has a real, non-fabricated chart."""
    preprocessor = pipe.named_steps["preprocessor"]
    clf = pipe.named_steps["classifier"]
    feat_names = prettify_feature_names(get_feature_names_out(preprocessor))

    if hasattr(clf, "feature_importances_"):
        importances = clf.feature_importances_
    elif hasattr(clf, "coef_"):
        importances = np.abs(clf.coef_[0])
    else:
        print("No native importances_/coef_ — falling back to permutation importance.")
        X_test_transformed = preprocessor.transform(X_test)
        if hasattr(X_test_transformed, "toarray"):
            X_test_transformed = X_test_transformed.toarray()
        result = permutation_importance(
            clf, X_test_transformed, y_test, n_repeats=20, random_state=RANDOM_STATE, n_jobs=1
        )
        importances = result.importances_mean

    order = np.argsort(importances)[::-1][:top_n]
    plt.figure(figsize=(7, 6))
    sns.barplot(
        x=importances[order],
        y=[feat_names[i] for i in order],
        hue=[feat_names[i] for i in order],
        palette="viridis",
        legend=False,
    )
    plt.title("Feature Importance (final model)")
    plt.xlabel("Importance")
    plt.tight_layout()
    plt.savefig(path, dpi=120)
    plt.close()


def plot_calibration_curve(y_test, proba_before, proba_after, path):
    plt.figure(figsize=(5.5, 5))
    for label, proba in [("Uncalibrated", proba_before), ("Calibrated", proba_after)]:
        frac_pos, mean_pred = calibration_curve(y_test, proba, n_bins=8, strategy="quantile")
        plt.plot(mean_pred, frac_pos, marker="o", label=label)
    plt.plot([0, 1], [0, 1], linestyle="--", color="gray", label="Perfectly calibrated")
    plt.xlabel("Mean predicted probability")
    plt.ylabel("Fraction of positives (actual)")
    plt.title("Calibration Curve — Before vs After")
    plt.legend()
    plt.tight_layout()
    plt.savefig(path, dpi=120)
    plt.close()


def plot_shap_summary(pipe, X_test_raw, path, max_display=15):
    """Point 3: explainable AI. Uses a model-agnostic-but-fast TreeExplainer
    path when possible, otherwise falls back to a general Explainer."""
    preprocessor = pipe.named_steps["preprocessor"]
    clf = pipe.named_steps["classifier"]
    X_test_transformed = preprocessor.transform(X_test_raw)
    if hasattr(X_test_transformed, "toarray"):
        X_test_transformed = X_test_transformed.toarray()
    feat_names = prettify_feature_names(get_feature_names_out(preprocessor))

    try:
        explainer = shap.TreeExplainer(clf)
        shap_values = explainer.shap_values(X_test_transformed)
        if isinstance(shap_values, list):  # older SHAP API: per-class list
            shap_values = shap_values[1]
        elif isinstance(shap_values, np.ndarray) and shap_values.ndim == 3:
            # newer SHAP API: shape (n_samples, n_features, n_classes)
            shap_values = shap_values[:, :, 1]
    except Exception:
        # Generic fallback for non-tree models (SVM, KNN, Logistic Regression)
        background = shap.sample(X_test_transformed, min(50, len(X_test_transformed)))
        explainer = shap.KernelExplainer(clf.predict_proba, background)
        shap_values = explainer.shap_values(X_test_transformed[:60], nsamples=100)
        if isinstance(shap_values, list):
            shap_values = shap_values[1]
        elif isinstance(shap_values, np.ndarray) and shap_values.ndim == 3:
            shap_values = shap_values[:, :, 1]
        X_test_transformed = X_test_transformed[:60]

    plt.figure()
    shap.summary_plot(
        shap_values,
        X_test_transformed,
        feature_names=feat_names,
        max_display=max_display,
        show=False,
    )
    plt.tight_layout()
    plt.savefig(path, dpi=120, bbox_inches="tight")
    plt.close()
    return explainer, feat_names


def error_analysis(X_test_raw, y_test, y_pred, y_proba) -> dict:
    """Point 7: where does the model fail, and what do those patients have
    in common? Focused especially on false negatives (missed disease) since
    that's the costlier error in a screening context."""
    df = X_test_raw.copy()
    df["actual"] = y_test.values
    df["predicted"] = y_pred
    df["probability"] = y_proba

    fn = df[(df["actual"] == 1) & (df["predicted"] == 0)]
    fp = df[(df["actual"] == 0) & (df["predicted"] == 1)]
    tp = df[(df["actual"] == 1) & (df["predicted"] == 1)]
    tn = df[(df["actual"] == 0) & (df["predicted"] == 0)]

    summary = {
        "n_test": len(df),
        "false_negatives": len(fn),
        "false_positives": len(fp),
        "false_negative_rate_of_positives": (
            len(fn) / max(1, len(fn) + len(tp))
        ),
        "false_positive_rate_of_negatives": (
            len(fp) / max(1, len(fp) + len(tn))
        ),
    }

    if len(fn) > 0:
        summary["false_negative_profile_mean"] = (
            fn[NUMERIC_FEATURES].mean().round(1).to_dict()
        )
        summary["true_positive_profile_mean"] = (
            tp[NUMERIC_FEATURES].mean().round(1).to_dict()
        )
        summary["false_negative_mean_probability"] = float(fn["probability"].mean())

    return summary


def main():
    print("=" * 70)
    print("HeartWise training pipeline")
    print("=" * 70)

    df = load_dataset()
    X = df[ALL_FEATURES]
    y = df[TARGET_COLUMN]

    plot_class_distribution(y, FIGURES_DIR / "class_distribution.png")

    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.2, stratify=y, random_state=RANDOM_STATE
    )
    print(f"\nDataset: {len(df)} patients ({y.mean():.1%} positive class)")
    print(f"Train: {len(X_train)}   Test: {len(X_test)}")

    # ---- Step 0: data-driven monotonic constraints for XGBoost. Fit a
    # throwaway preprocessor just to learn the exact post-encoding column
    # order (deterministic — the real pipelines below will reproduce it
    # identically when fit on the same X_train). ----
    from data_pipeline import build_preprocessor

    _throwaway_preprocessor = build_preprocessor().fit(X_train)
    _feature_names_raw = get_feature_names_out(_throwaway_preprocessor)
    monotone_constraints, monotone_directions = compute_monotone_constraints(
        X_train, y_train, _feature_names_raw
    )
    print("\n--- Monotonic constraints for XGBoost (data-driven) ---")
    for feat, (sign, corr) in monotone_directions.items():
        arrow = "increasing" if sign > 0 else "decreasing"
        print(f"  {FEATURE_LABELS.get(feat, feat):40s} {arrow:11s} (train corr={corr:+.3f})")
    unconstrained = set(MONOTONE_CANDIDATE_FEATURES) - set(monotone_directions)
    if unconstrained:
        print(f"  (left unconstrained — correlation below {MONOTONE_CORR_THRESHOLD}: "
              f"{', '.join(FEATURE_LABELS.get(f, f) for f in unconstrained)})")

    # ---- Step 1: Stratified 5-fold CV comparison across 6 models ----
    print("\n--- Stratified 5-Fold Cross-Validation ---")
    cv_results = cross_validate_models(X_train, y_train, monotone_constraints)

    # ---- Step 2: fit each model on the full training set and score on the
    # held-out test set (a second, independent check beyond CV) ----
    print("\n--- Held-out test set evaluation ---")
    fitted_pipes = {}
    test_rows = []
    for name, model in get_candidate_models(monotone_constraints).items():
        pipe = make_pipeline(model)
        pipe.fit(X_train, y_train)
        fitted_pipes[name] = pipe
        row = evaluate_on_test(name, pipe, X_test, y_test)
        test_rows.append(row)
        print(
            f"[TEST] {name:22s} Acc={row['test_accuracy']:.3f}  "
            f"Prec={row['test_precision']:.3f}  Rec={row['test_recall']:.3f}  "
            f"F1={row['test_f1']:.3f}  ROC-AUC={row['test_roc_auc']:.3f}"
        )
    test_results = pd.DataFrame(test_rows)

    comparison = cv_results.merge(test_results, on="model")
    comparison.to_csv(REPORTS_DIR / "model_comparison.csv", index=False)

    # ---- Step 3: pick the winner. For a screening tool, recall and ROC-AUC
    # matter more than raw accuracy (a missed positive is costlier than a
    # false alarm) — see improvement note #2. We rank on CV ROC-AUC first,
    # recall as a tiebreaker. Then two further, explicitly documented
    # tie-breaks apply, in order:
    #   (a) explainability — a tree-based model (fast, exact SHAP + native
    #       feature importance) is preferred over a black-box-ish model
    #       like KNN/SVM if it's within a small tolerance of the top score.
    #   (b) prediction stability — among tree-based candidates, the
    #       monotonic-constrained XGBoost is preferred if it's competitive,
    #       because it's the only one immune to the small-sample zig-zags
    #       (see compute_monotone_constraints) that a plain Random Forest
    #       can produce on sparse values like ca=4 (n=5 in the whole
    #       dataset). This is a real reliability requirement for a tool
    #       people will interact with, not just a metrics contest.
    comparison["rank_score"] = comparison["roc_auc_mean"] * 0.7 + comparison["recall_mean"] * 0.3
    comparison_sorted = comparison.sort_values("rank_score", ascending=False).reset_index(drop=True)
    top_score = comparison_sorted.iloc[0]["rank_score"]
    top_name = comparison_sorted.iloc[0]["model"]

    TREE_BASED = {"Random Forest", "XGBoost", "Decision Tree"}
    INTERPRETABILITY_TOLERANCE = 0.015
    STABILITY_TOLERANCE = 0.03
    within_interpretability = comparison_sorted[
        comparison_sorted["rank_score"] >= top_score - INTERPRETABILITY_TOLERANCE
    ]
    tree_candidates = within_interpretability[within_interpretability["model"].isin(TREE_BASED)]
    xgb_row = comparison_sorted[comparison_sorted["model"] == "XGBoost"].iloc[0]

    if xgb_row["rank_score"] >= top_score - STABILITY_TOLERANCE:
        best_row = xgb_row
        top_tree_alt = comparison_sorted[
            comparison_sorted["model"].isin(TREE_BASED - {"XGBoost"})
        ].sort_values("rank_score", ascending=False).iloc[0]
        selection_note = (
            f"XGBoost, with monotonic constraints on "
            f"{', '.join(FEATURE_LABELS.get(f, f) for f in monotone_directions)} "
            f"(directions measured from training-data correlations, not assumed), was preferred "
            f"as the production model. Its CV rank-score ({xgb_row['rank_score']:.4f}) is within "
            f"{STABILITY_TOLERANCE} of the top performer ({top_name}, {top_score:.4f}), and its "
            f"recall ({xgb_row['recall_mean']:.3f}) is at least as good as the strongest "
            f"alternative tree-based model, {top_tree_alt['model']} ({top_tree_alt['recall_mean']:.3f}). "
            "Unlike Random Forest — which, even after removing 7 rows of almost-certainly-"
            "placeholder missing-data codes (ca=4, thal=0), still showed small non-monotonic "
            "wiggles when we swept oldpeak and max heart rate across their range — XGBoost's "
            "constraints make that class of artifact structurally impossible, not just less "
            "likely, while keeping the same fast, exact SHAP explanations every tree-based model "
            "here gets."
        )
    elif not tree_candidates.empty and top_name not in TREE_BASED:
        best_row = tree_candidates.sort_values("rank_score", ascending=False).iloc[0]
        selection_note = (
            f"{best_row['model']} scored within {INTERPRETABILITY_TOLERANCE} rank-score points "
            f"of the top performer ({top_name}, {top_score:.4f}) and was preferred for its fast, "
            "exact SHAP explanations and native feature importance — important for a healthcare "
            "screening tool where every prediction needs to be explainable, not just accurate."
        )
    else:
        best_row = comparison_sorted.iloc[0]
        selection_note = (
            "Selected as the top scorer on Stratified 5-Fold CV "
            "(rank_score = 0.7 x ROC-AUC + 0.3 x Recall)."
        )
    best_name = best_row["model"]
    best_pipe = fitted_pipes[best_name]
    print(f"\n>>> Selected model: {best_name} "
          f"(CV ROC-AUC={best_row['roc_auc_mean']:.3f}, CV Recall={best_row['recall_mean']:.3f})")
    print(f">>> Rationale: {selection_note}")

    plot_roc_curves(fitted_pipes, X_test, y_test, FIGURES_DIR / "roc_curves.png")

    y_pred_best = best_pipe.predict(X_test)
    y_proba_best_uncalibrated = best_pipe.predict_proba(X_test)[:, 1]
    plot_confusion_matrix(y_test, y_pred_best, best_name, FIGURES_DIR / "confusion_matrix.png")
    plot_feature_importance(best_pipe, X_test, y_test, FIGURES_DIR / "feature_importance.png")

    # ---- Step 4: probability calibration (point 5) ----
    print("\n--- Probability calibration ---")
    brier_uncalibrated = brier_score_loss(y_test, y_proba_best_uncalibrated)

    cv_cal = StratifiedKFold(n_splits=5, shuffle=True, random_state=RANDOM_STATE)
    sigmoid_cal = CalibratedClassifierCV(
        make_pipeline(get_candidate_models(monotone_constraints)[best_name]), method="sigmoid", cv=cv_cal
    )
    sigmoid_cal.fit(X_train, y_train)
    proba_sigmoid = sigmoid_cal.predict_proba(X_test)[:, 1]
    brier_sigmoid = brier_score_loss(y_test, proba_sigmoid)

    isotonic_cal = CalibratedClassifierCV(
        make_pipeline(get_candidate_models(monotone_constraints)[best_name]), method="isotonic", cv=cv_cal
    )
    isotonic_cal.fit(X_train, y_train)
    proba_isotonic = isotonic_cal.predict_proba(X_test)[:, 1]
    brier_isotonic = brier_score_loss(y_test, proba_isotonic)

    print(f"Brier score — uncalibrated: {brier_uncalibrated:.4f}")
    print(f"Brier score — Platt/sigmoid: {brier_sigmoid:.4f}")
    print(f"Brier score — isotonic:      {brier_isotonic:.4f}")

    calibration_options = {
        "none": (brier_uncalibrated, best_pipe, y_proba_best_uncalibrated),
        "sigmoid": (brier_sigmoid, sigmoid_cal, proba_sigmoid),
        "isotonic": (brier_isotonic, isotonic_cal, proba_isotonic),
    }
    best_cal_method = min(calibration_options, key=lambda k: calibration_options[k][0])
    best_brier, final_model, final_proba = calibration_options[best_cal_method]
    print(f">>> Best calibration method: {best_cal_method} (Brier={best_brier:.4f})")

    plot_calibration_curve(
        y_test, y_proba_best_uncalibrated, final_proba, FIGURES_DIR / "calibration_curve.png"
    )

    # ---- Step 5: SHAP explainability (point 3) — always computed on the
    # *uncalibrated* base pipeline, since that's what has a real tree/coef
    # structure to explain; the calibration layer only rescales its output. ----
    print("\n--- SHAP explainability ---")
    try:
        plot_shap_summary(best_pipe, X_test, FIGURES_DIR / "shap_summary.png")
        print("SHAP summary plot saved.")
    except Exception as e:
        print(f"SHAP summary plot skipped due to: {e}")

    # ---- Step 6: error analysis (point 7) ----
    print("\n--- Error analysis ---")
    final_pred = (final_proba >= 0.5).astype(int)
    err = error_analysis(X_test, y_test, final_pred, final_proba)
    print(json.dumps(err, indent=2, default=str))
    with open(REPORTS_DIR / "error_analysis.json", "w") as f:
        json.dump(err, f, indent=2, default=str)

    # ---- Step 7: save everything (point 12 — model versioning) ----
    model_path = MODELS_DIR / "heart_model_pipeline.joblib"
    joblib.dump(final_model, model_path)
    # also stash the uncalibrated base pipeline for fast SHAP explanations at
    # inference time, and the fitted calibrated model as the one that's
    # actually used for the displayed probability
    joblib.dump(best_pipe, MODELS_DIR / "heart_model_base_uncalibrated.joblib")

    final_metrics = {
        "model_family": best_name,
        "calibration_method": best_cal_method,
        "test_accuracy": float(accuracy_score(y_test, final_pred)),
        "test_precision": float(precision_score(y_test, final_pred)),
        "test_recall": float(recall_score(y_test, final_pred)),
        "test_f1": float(f1_score(y_test, final_pred)),
        "test_roc_auc": float(roc_auc_score(y_test, final_proba)),
        "brier_score": float(best_brier),
        "cv_roc_auc_mean": float(best_row["roc_auc_mean"]),
        "cv_roc_auc_std": float(best_row["roc_auc_std"]),
        "cv_recall_mean": float(best_row["recall_mean"]),
        "cv_recall_std": float(best_row["recall_std"]),
    }
    with open(REPORTS_DIR / "final_metrics.json", "w") as f:
        json.dump(final_metrics, f, indent=2)

    metadata = {
        "model_name": "HeartWise-RiskModel",
        "version": MODEL_VERSION,
        "model_family": best_name,
        "calibration_method": best_cal_method,
        "selection_rationale": selection_note,
        "monotone_constraints": {
            FEATURE_LABELS.get(feat, feat): ("increasing" if sign > 0 else "decreasing")
            for feat, (sign, _corr) in monotone_directions.items()
        } if best_name == "XGBoost" else {},
        "training_date_utc": datetime.now(timezone.utc).isoformat(),
        "dataset": {
            "name": "UCI Heart Disease (Cleveland, processed)",
            "n_records": int(len(df)),
            "n_train": int(len(X_train)),
            "n_test": int(len(X_test)),
            "positive_rate": float(y.mean()),
        },
        "features": ALL_FEATURES,
        "numeric_features": NUMERIC_FEATURES,
        "categorical_features": CATEGORICAL_FEATURES,
        "feature_labels": FEATURE_LABELS,
        "metrics": final_metrics,
        "decision_threshold": 0.5,
        "intended_use": (
            "Educational / research decision-support demo only. "
            "Not a certified medical device and must not be used for actual "
            "clinical diagnosis."
        ),
    }
    with open(MODELS_DIR / "metadata.json", "w") as f:
        json.dump(metadata, f, indent=2)

    print(f"\nSaved model      -> {model_path}")
    print(f"Saved metadata   -> {MODELS_DIR / 'metadata.json'}")
    print(f"Saved comparison -> {REPORTS_DIR / 'model_comparison.csv'}")
    print(f"Saved figures    -> {FIGURES_DIR}")
    print("\nDone.")
