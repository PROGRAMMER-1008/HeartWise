def risk_css_class(risk_level: str) -> str:
    return {"Low": "low", "Borderline": "borderline", "Elevated": "elevated"}.get(risk_level, "low")

# ---------------------------------------------------------------------------
# Model loading (in-process fallback path)
# ---------------------------------------------------------------------------
@st.cache_resource
def load_local_explainer():
    from explain import get_explainer

    return get_explainer()


@st.cache_data
def load_metadata():
    with open(MODELS_DIR / "metadata.json") as f:
        return json.load(f)


def api_is_up() -> bool:
    try:
        r = requests.get(f"{API_URL}/health", timeout=1.0)
        return r.status_code == 200
    except requests.RequestException:
        return False


def run_prediction(patient: dict) -> dict:
    """Try the FastAPI backend first; fall back to the in-process model."""
    if api_is_up():
        try:
            r = requests.post(f"{API_URL}/predict", json=patient, timeout=5.0)
            r.raise_for_status()
            return r.json(), "api"
        except requests.RequestException:
            pass
    explainer = load_local_explainer()
    result = explainer.predict(patient)
    metadata = load_metadata()
    result["model_version"] = metadata["version"]
    return result, "local"
