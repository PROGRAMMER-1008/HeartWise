"""
report.py
---------
Builds the downloadable "Patient Risk Report" PDF (improvement note #10).
Kept separate from the UI code so both Streamlit and (if ever needed) the
API can generate the same report.
"""

import io
import sys
from datetime import datetime
from pathlib import Path

from reportlab.lib import colors
from reportlab.lib.pagesizes import A4
from reportlab.lib.styles import ParagraphStyle, getSampleStyleSheet
from reportlab.lib.units import cm
from reportlab.platypus import (
    Paragraph,
    SimpleDocTemplate,
    Spacer,
    Table,
    TableStyle,
)

sys.path.insert(0, str(Path(__file__).resolve().parent))
from data_pipeline import CATEGORY_OPTIONS, FEATURE_LABELS  # noqa: E402

DISCLAIMER = (
    "This report is generated by an educational / research machine-learning "
    "demo. It is NOT a medical diagnosis and must not be used to make real "
    "clinical decisions. Please consult a qualified healthcare professional "
    "for any genuine health concern."
)


def _format_patient_value(field: str, value) -> str:
    if field in CATEGORY_OPTIONS:
        return str(CATEGORY_OPTIONS[field].get(int(value), value))
    return str(value)


def build_pdf_report(patient: dict, result: dict, model_version: str) -> bytes:
    """Returns the PDF file content as bytes (so Streamlit can offer it as a
    download without ever writing to disk)."""
    buffer = io.BytesIO()
    doc = SimpleDocTemplate(
        buffer,
        pagesize=A4,
        topMargin=1.5 * cm,
        bottomMargin=1.5 * cm,
        leftMargin=2 * cm,
        rightMargin=2 * cm,
    )
    styles = getSampleStyleSheet()
    title_style = ParagraphStyle(
        "TitleStyle", parent=styles["Title"], textColor=colors.HexColor("#B3261E")
    )
    section_style = ParagraphStyle(
        "SectionStyle", parent=styles["Heading2"], spaceBefore=12, spaceAfter=6
    )
    body = styles["BodyText"]
    small = ParagraphStyle("Small", parent=styles["BodyText"], fontSize=8, textColor=colors.grey)

    elements = []
    elements.append(Paragraph("HEARTWISE — Heart Risk Report", title_style))
    elements.append(
        Paragraph(
            f"Patient ID: {patient.get('patient_id', 'N/A')} &nbsp;&nbsp;|&nbsp;&nbsp; "
            f"Generated: {datetime.now().strftime('%Y-%m-%d %H:%M')} &nbsp;&nbsp;|&nbsp;&nbsp; "
            f"Model version: {model_version}",
            body,
        )
    )
    elements.append(Spacer(1, 12))

    # --- Result summary ---
    elements.append(Paragraph("Model Output", section_style))
    risk_color = {
        "Low": colors.HexColor("#2E7D32"),
        "Borderline": colors.HexColor("#B8860B"),
        "Elevated": colors.HexColor("#B3261E"),
    }.get(result["risk_level"], colors.black)
    result_table = Table(
        [
            ["Prediction", result["prediction_label"]],
            ["Risk band", result["risk_level"]],
            ["Model probability (positive class)", f"{result['probability']:.2f}"],
        ],
        colWidths=[7 * cm, 7 * cm],
    )
    result_table.setStyle(
        TableStyle(
            [
                ("BACKGROUND", (0, 0), (-1, 0), colors.whitesmoke),
                ("TEXTCOLOR", (1, 1), (1, 1), risk_color),
                ("FONTNAME", (1, 1), (1, 1), "Helvetica-Bold"),
                ("GRID", (0, 0), (-1, -1), 0.5, colors.lightgrey),
                ("FONTSIZE", (0, 0), (-1, -1), 10),
                ("TOPPADDING", (0, 0), (-1, -1), 6),
                ("BOTTOMPADDING", (0, 0), (-1, -1), 6),
            ]
        )
    )
    elements.append(result_table)
    elements.append(
        Paragraph(
            "Note: this is the model's raw output probability, not a clinically validated "
            "probability of disease — see the calibration note in the dashboard for why that "
            "distinction matters.",
            small,
        )
    )
    elements.append(Spacer(1, 10))

    # --- Top factors ---
    elements.append(Paragraph("Why the model reached this result", section_style))
    if result.get("top_factors"):
        factor_rows = [["Factor", "Effect"]]
        for f in result["top_factors"]:
            arrow = "▲ increases risk" if f["direction"] == "increases risk" else "▼ decreases risk"
            factor_rows.append([f["feature"], arrow])
        factor_table = Table(factor_rows, colWidths=[9 * cm, 5 * cm])
        factor_table.setStyle(
            TableStyle(
                [
                    ("BACKGROUND", (0, 0), (-1, 0), colors.HexColor("#F0F0F0")),
                    ("FONTNAME", (0, 0), (-1, 0), "Helvetica-Bold"),
                    ("GRID", (0, 0), (-1, -1), 0.5, colors.lightgrey),
                    ("FONTSIZE", (0, 0), (-1, -1), 9.5),
                    ("TOPPADDING", (0, 0), (-1, -1), 5),
                    ("BOTTOMPADDING", (0, 0), (-1, -1), 5),
                ]
            )
        )
        elements.append(factor_table)
    else:
        elements.append(Paragraph("No per-patient explanation available for this model.", body))
    elements.append(Spacer(1, 10))

    # --- Patient inputs ---
    elements.append(Paragraph("Patient Information Used", section_style))
    info_rows = [["Field", "Value"]]
    for field, label in FEATURE_LABELS.items():
        if field in patient:
            info_rows.append([label, _format_patient_value(field, patient[field])])
    info_table = Table(info_rows, colWidths=[9 * cm, 5 * cm])
    info_table.setStyle(
        TableStyle(
            [
                ("BACKGROUND", (0, 0), (-1, 0), colors.HexColor("#F0F0F0")),
                ("FONTNAME", (0, 0), (-1, 0), "Helvetica-Bold"),
                ("GRID", (0, 0), (-1, -1), 0.5, colors.lightgrey),
                ("FONTSIZE", (0, 0), (-1, -1), 9.5),
                ("TOPPADDING", (0, 0), (-1, -1), 4),
                ("BOTTOMPADDING", (0, 0), (-1, -1), 4),
            ]
        )
    )
    elements.append(info_table)
    elements.append(Spacer(1, 16))

    # --- Disclaimer ---
    elements.append(Paragraph(DISCLAIMER, small))

    doc.build(elements)
    return buffer.getvalue()
